/* =========================================================
KONFIGURASI API
========================================================= */

const API_URL =
"https://script.google.com/macros/s/AKfycbz5TsgImoG_XGxtNurLTZtJBRiYx3iOhQ0XxfhTVBVVsUB8C_R7Yfh6xfUofbgpTsvY/exec";


/* =========================================================
STATE
========================================================= */

let currentUser = null;

let masterSiswa = [];

let currentSiswa = [];

let currentPage = "dashboard";


/* =========================================================
API
========================================================= */

async function api(action, params = {}) {

    const query =
        new URLSearchParams();

    query.append(
        "action",
        action
    );

    Object.keys(params).forEach(key => {

        query.append(
            key,
            params[key] ?? ""
        );

    });


    const response =
        await fetch(
            API_URL +
            "?" +
            query.toString()
        );


    return await response.json();

}


/* =========================================================
LOGIN
========================================================= */

document
.getElementById("loginForm")
.addEventListener(
    "submit",
    async function(e) {

        e.preventDefault();

        const username =
            document
            .getElementById("username")
            .value;

        const password =
            document
            .getElementById("password")
            .value;


        const msg =
            document
            .getElementById("loginMessage");


        msg.innerHTML =
            "⏳ Memeriksa login...";


        try {

            const result =
                await api(
                    "login",
                    {
                        username,
                        password
                    }
                );


            if (
                result.status ===
                "sukses"
            ) {

                currentUser =
                    result.user;


                localStorage.setItem(
                    "piketUser",
                    JSON.stringify(
                        currentUser
                    )
                );


                masukAplikasi();

            } else {

                msg.innerHTML =
                    `<span class="text-red-600">
                    ${result.pesan}
                    </span>`;

            }

        } catch(error) {

            msg.innerHTML =
                `<span class="text-red-600">
                Gagal menghubungi server.
                </span>`;

        }

    }
);


/* =========================================================
MASUK APLIKASI
========================================================= */

function masukAplikasi() {

    document
    .getElementById("loginPage")
    .classList.add("hidden");


    document
    .getElementById("app")
    .classList.remove("hidden");


    document
    .getElementById("userName")
    .textContent =
        currentUser.nama;


    document
    .getElementById("userRole")
    .textContent =
        currentUser.role;


    if (
        currentUser.role !==
        "ADMIN"
    ) {

        document
        .getElementById("menuUsers")
        .style.display =
            "none";

    }


    setTanggalHariIni();

    loadDashboard();

    loadSiswa();

}


/* =========================================================
CEK SESSION
========================================================= */

function checkSession() {

    const saved =
        localStorage.getItem(
            "piketUser"
        );


    if (saved) {

        try {

            currentUser =
                JSON.parse(saved);

            masukAplikasi();

        } catch(e) {

            localStorage.removeItem(
                "piketUser"
            );

        }

    }

}


/* =========================================================
LOGOUT
========================================================= */

function logout() {

    localStorage.removeItem(
        "piketUser"
    );

    location.reload();

}


/* =========================================================
PAGE NAVIGATION
========================================================= */

function showPage(page) {

    currentPage = page;


    document
    .querySelectorAll(
        "[id^='page-']"
    )
    .forEach(el => {

        el.classList.add(
            "hidden-page"
        );

    });


    const target =
        document
        .getElementById(
            "page-" + page
        );


    if (target) {

        target.classList.remove(
            "hidden-page"
        );

    }


    document
    .querySelectorAll(
        ".menu-item"
    )
    .forEach(el => {

        el.classList.remove(
            "active"
        );

    });


    const menu =
        document
        .querySelector(
            `[data-menu="${page}"]`
        );


    if (menu) {

        menu.classList.add(
            "active"
        );

    }


    const titles = {

        dashboard:
            "Dashboard",

        input:
            "Input Keterlambatan",

        laporan:
            "Rekap Keterlambatan",

        siswa:
            "Data Siswa",

        users:
            "Manajemen User"

    };


    document
    .getElementById("pageTitle")
    .textContent =
        titles[page] || "Dashboard";


    if (page === "dashboard")
        loadDashboard();

    if (page === "laporan")
        loadLaporan();

    if (page === "users")
        loadUsers();

function filterSiswa() {

    const search =
        document
        .getElementById(
            "searchSiswa"
        )
        .value
        .toLowerCase();


    currentSiswa =
        masterSiswa.filter(
            siswa =>
                String(siswa.nama)

                String(siswa.nisn)
                .toLowerCase()
                .includes(search)
                ||
                String(siswa.kelas)
                .toLowerCase()
                .includes(search)
        );


    renderSiswaSelect();

}

.addEventListener(
    "input",
    updateDurasi

    "input",
    updateDurasi

function updateDurasi() {

    const masuk =
        document
        .getElementById(
            "jamMasuk"
        ).value;

    const batas =
        document
        .getElementById(
            "jamBatas"
        ).value;


    if (!masuk || !batas) {

        document
        .getElementById(
            "previewDurasi"
        )
        .textContent =
            "Durasi keterlambatan: 0 menit";

        return;

    }


    const m =
        masuk.split(":");

    const b =
        batas.split(":");


    const totalMasuk =
        Number(m[0]) * 60 +
        Number(m[1]);


    const totalBatas =
        Number(b[0]) * 60 +
        Number(b[1]);


    let durasi =
        totalMasuk -
        totalBatas;


    if (durasi < 0)
        durasi = 0;


    document
    .getElementById(
        "previewDurasi"
    )
    .textContent =
        `Durasi keterlambatan: ${durasi} menit`;

}

/* =========================================================

========================================================= */

async function loadUsers() {

    if (
        currentUser.role !==
        "ADMIN"
    ) return;


    const users =
        await api(
            "users"
        );


    const tbody =
        document
        .getElementById(
            "usersTable"
        );


    tbody.innerHTML = "";


    users.forEach(
        user => {

            const statusClass =
                user.status === "Aktif"
                    ? "bg-green-100 text-green-700"
                    : "bg-gray-100 text-gray-600";


            tbody.innerHTML += `

            <tr class="border-b">

                <td class="p-4 font-semibold">
                    ${user.username}
                </td>

                <td class="p-4">
                    ${user.nama}
                </td>

                <td class="p-4">
                    ${user.role}
                </td>

                <td class="p-4">

                    <span class="px-3 py-1
                                 rounded-full
                                 ${statusClass}">

                        ${user.status}

                    </span>

                </td>

                <td class="p-4">
                    ${user.lastLogin}
                </td>

                <td class="p-4 flex gap-2">

                    <button
                        onclick="toggleUser('${user.id}','${user.status}')"
                        class="px-3 py-2
                               bg-gray-100
                               rounded-lg">

                        ${user.status === "Aktif"
                            ? "Nonaktifkan"
                            : "Aktifkan"}

                    </button>


                    <button
                        onclick="deleteUser('${user.id}')"
                        class="px-3 py-2
                               bg-red-100
                               text-red-700
                               rounded-lg">

                        Hapus

                    </button>

                </td>

            </tr>

            `;

        }
    );

}


/* =========================================================
TOGGLE USER
========================================================= */

async function toggleUser(
    id,
    status
) {

    const newStatus =
        status === "Aktif"
            ? "Nonaktif"
            : "Aktif";


    if (
        !confirm(
            `Ubah status menjadi ${newStatus}?`
        )
    ) return;


    const result =
        await api(
            "ubahStatusUser",
            {
                id,
                status: newStatus
            }
        );


    alert(
        result.pesan
    );


    loadUsers();

}

/* =========================================================

function setTanggalHariIni() {

    const now =
        new Date();


    const local =
        new Date(
            now.getTime() -
            now.getTimezoneOffset()
            * 60000
        );


    document
    .getElementById(
        "tanggal"
    )
    .value =
        local
        .toISOString()
        .slice(0,10);

}
